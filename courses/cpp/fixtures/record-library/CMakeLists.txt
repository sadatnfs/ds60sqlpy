cmake_minimum_required(VERSION 3.28)
project(ds60_record_library VERSION 1.0.0 LANGUAGES CXX)
include(CTest)
add_library(ds60_records STATIC record.cpp)
target_include_directories(ds60_records PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>" "$<INSTALL_INTERFACE:include>")
target_compile_features(ds60_records PUBLIC cxx_std_20)
set_target_properties(ds60_records PROPERTIES CXX_EXTENSIONS OFF)
add_executable(record_consumer consumer.cpp)
target_link_libraries(record_consumer PRIVATE ds60_records)
add_test(NAME record_contract COMMAND record_consumer)
set_tests_properties(record_contract PROPERTIES TIMEOUT 5)
install(TARGETS ds60_records EXPORT ds60RecordTargets ARCHIVE DESTINATION lib)
install(FILES record.hpp DESTINATION include)
install(EXPORT ds60RecordTargets NAMESPACE ds60:: DESTINATION lib/cmake/ds60Records)

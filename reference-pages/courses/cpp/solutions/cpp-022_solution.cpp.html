std::string encode(int id,int cents){return "v1,"+std::to_string(id)+","+std::to_string(cents);}
struct DecodedRecord { int id; int cents; };
std::optional<DecodedRecord> decode(std::string_view text) {
    if (!text.starts_with("v1,")) return {};
    text.remove_prefix(3);
    const auto comma=text.find(',');
    if (comma==text.npos || text.find(',',comma+1)!=text.npos) return {};
    int id=0,cents=0;
    auto [id_end,id_error]=std::from_chars(text.data(),text.data()+comma,id);
    auto [amount_end,amount_error]=std::from_chars(text.data()+comma+1,text.data()+text.size(),cents);
    if (id_error!=std::errc{} || amount_error!=std::errc{} || id_end!=text.data()+comma
        || amount_end!=text.data()+text.size() || id<0 || cents<0) return {};
    return DecodedRecord{id,cents};
}
